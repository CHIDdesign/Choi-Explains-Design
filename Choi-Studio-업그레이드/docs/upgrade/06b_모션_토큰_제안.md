## 1. `motion.ts` — 바꿀 블록

`EASE` 와 `SPRING` 은 블록째 바꾼다(기존 키 유지 + 새 키). 나머지는 파일 끝에 붙인다.

```ts
// 이징 — 곡선 이름이 아니라 '역할'로 고른다(v2).
// 진입·이동·퇴장에 서로 다른 곡선을 쓴다. 같은 곡선을 전부에 쓰면 모든 요소가 같은 성격으로 움직인다.
export const EASE = {
  // ── 기존 키(그대로 둔다) ──
  outQuint: Easing.bezier(0.23, 1, 0.32, 1),
  outExpo: Easing.bezier(0.19, 1, 0.22, 1),
  outCubic: Easing.bezier(0.33, 1, 0.68, 1),
  inOutCubic: Easing.bezier(0.645, 0.045, 0.355, 1),
  inCubic: Easing.bezier(0.32, 0, 0.67, 0),
  outBack: Easing.bezier(0.34, 1.56, 0.64, 1),
  inExpo: Easing.bezier(0.7, 0, 0.84, 0),
  inQuart: Easing.bezier(0.5, 0, 0.75, 0),
  linear: Easing.linear,
  // ── v2 역할 이름 ──
  enter: Easing.bezier(0.22, 1, 0.36, 1),          // 작은·중간 요소 진입 — easeOutQuint(easings.net)
  enterText: Easing.bezier(0.16, 1, 0.3, 1),       // 글 마스크 리빌 — easeOutExpo
  enterLarge: Easing.bezier(0.05, 0.7, 0.1, 1),    // 큰 면(종이 판·전면) — Material 3 emphasized-decelerate
  move: Easing.bezier(0.2, 0, 0, 1),               // 화면 안 A→B — Material 3 standard(비대칭: 빨리 떠나 길게 앉는다)
  exit: Easing.bezier(0.3, 0, 0.8, 0.15),          // 퇴장 — Material 3 emphasized-accelerate
  settle: Easing.bezier(0.34, 1.2, 0.64, 1),       // 미세 정착, 피크 약 +1.3% — 숫자 끝·칩

  pop: Easing.bezier(0.34, 1.56, 0.64, 1),         // 피크 약 +9.8% — 펀치 구간·숏폼 훅 전용(롱폼 본문 금지)
} as const;

// 스프링 — 감쇠비 ζ = damping / (2·√(stiffness·mass)). 오버슛 = exp(−ζπ/√(1−ζ²)).
// Remotion 기본값(100/10)은 ζ 0.5, 오버슛 약 16% 라 쓰지 않는다. 길이는 measureSpring 으로 잰다.
export const SPRING = {
  gentle: {stiffness: 100, damping: 20, mass: 1}, // ζ 1.0  · 0%
  stiff: {stiffness: 400, damping: 30, mass: 1},  // ζ 0.75 · 약 2.8%
  slow: {stiffness: 50, damping: 20, mass: 1},    // ζ 1.41 · 0%
  // ── v2 ──
  paper: {stiffness: 260, damping: 22, mass: 1},  // ζ 0.68 · 약 5% — 메모의 '회전' 정착에만(크기·위치에는 쓰지 않는다)
  snap: {stiffness: 500, damping: 32, mass: 1},   // ζ 0.72 · 약 4% — 칩·배지·테이프
  heavy: {stiffness: 120, damping: 22, mass: 1},  // ζ 1.0  · 0%  — 종이 판·전면
```

## 2. `motion.ts` — 붙일 블록

```ts
/** 퇴장은 진입보다 짧다. 기본 0.75, 허용 0.6~0.87(NN/g 300→200~250ms, Material 225→195ms). */
export const EXIT_RATIO = 0.75;

/**
 * 길이는 거리와 면적으로 고른다(30fps 프레임).
 * distPct: 이동 거리(프레임 짧은 변 대비 %), areaPct: 요소 면적(프레임 대비 %)
 */
export const durFor = (distPct: number, areaPct = 0): number => {
  const d = distPct < 5 ? 8 : distPct < 15 ? 11 : distPct < 35 ? 15 : distPct < 60 ? 19 : 23;
  return Math.min(26, d + (areaPct > 50 ? 3 : areaPct > 25 ? 2 : 0));
};

/** 장식적 스태거의 총 길이 상한 15f(0.5초). 말에 맞춘 순차 등장(목록·단계)에는 쓰지 않는다. */
export const staggerFor = (n: number, base: number, cap = 15): number =>
  n <= 1 ? 0 : Math.max(1, Math.min(base, Math.floor(cap / (n - 1))));

/** 읽는 시간: 모든 모션이 끝난 뒤 글자가 멈춰 있어야 하는 길이. 7자/초(공백·부호 제외). */
export const HOLD = {cps: 7, minSec: 1.2} as const;
export const restFrames = (chars: number, fps = 30): number =>
  Math.ceil(Math.max(HOLD.minSec, chars / HOLD.cps) * fps);

/** 부품 시간차(팔로스루): 본체 도착 기준 프레임. 꼬리는 maxTail 안에서 끝난다. */
export const FOLLOW = {shadow: 2, tape: 3, badge: 4, underline: 6, note: 8, maxTail: 12} as const;

/** 진입·퇴장 가족 — 변주 선택기가 고른다(06 문서 5·6장). */
export const ENTER_KINDS = ['place', 'slide', 'unfold', 'rule'] as const;
export const EXIT_KINDS = ['fade', 'peel', 'swap'] as const;
export type EnterKind = (typeof ENTER_KINDS)[number];
export type ExitKind = (typeof EXIT_KINDS)[number];

/**
 * 손맛 스텝: hold 프레임마다만 값이 바뀐다(2 = 30fps 에서 15fps, 4 = 7.5fps).
 * 종이·테이프·사진·손글씨에만 쓴다. 자막·화자 영상·카메라에는 쓰지 않는다. 20f 넘는 이동에도 쓰지 않는다.
 */
export const stepFrame = (frame: number, hold = 2): number => Math.floor(frame / hold) * hold;

/** 보일: 정지한 종이가 4프레임마다 아주 조금 다르게 놓인다. 글자는 종이와 같은 변환 안에 둔다. */
export const BOIL = {hold: 4, rotate: 0.4, shift: 0.8, seeds: 3} as const; // ±도, ±px
const rnd = (n: number): number => {
  const x = Math.sin(n * 127.1 + 311.7) * 43758.5453;
  return x - Math.floor(x);
};
export const boil = (frame: number, seed: number): {rotate: number; dx: number; dy: number} => {
  const k = Math.floor(frame / BOIL.hold) % BOIL.seeds;
  const r = (i: number) => rnd(seed * 13 + k * 7 + i) * 2 - 1;
  return {rotate: r(1) * BOIL.rotate, dx: r(2) * BOIL.shift, dy: r(3) * BOIL.shift};
};

/**
 * 붙이기(place): 종이가 살짝 떠 있다가 내려앉는다. 스케일 바운스가 아니라 회전과 그림자가 정착한다.
 * rest = 도착 각(시드로 정한 값). elev 는 tokens.ts paperShadow(elev) 에 넣는다(3 → 1).
 */
export const placeIn = (frame: number, rest: number, dur = 10, step = true) => {
  const f = step ? stepFrame(frame, 2) : frame;
  const p = interpolate(f, [0, dur], [0, 1], {...clamp, easing: EASE.enter});
  return {
    opacity: interpolate(f, [0, Math.max(2, Math.round(dur * 0.4))], [0, 1], clamp), // 불투명도는 먼저 끝난다
    scale: 1.03 - 0.03 * p,
    translateY: -14 * (1 - p),
    // 회전은 위치보다 3f 늦게 끝나고 도착 각을 0.5° 지나쳤다 돌아온다
    rotate: interpolate(f, [0, Math.round(dur * 0.7), dur + 3], [rest + 2.5, rest - 0.5, rest],
      {...clamp, easing: EASE.outCubic}),
    elev: 3 - 2 * p,
  };
};

/** 펼치기(unfold): 테이프가 붙은 변을 축으로 열린다. origin 은 그 변('top' | 'left' | 'right'). */
export const unfoldIn = (frame: number, dur = 12) => {
  const p = interpolate(frame, [0, dur], [0, 1], {...clamp, easing: EASE.enter});
  return {clip: 1 - p, scaleCross: 0.96 + 0.04 * p, opacity: interpolate(frame, [0, 4], [0, 1], clamp)};
};

/** 떼기(peel) 퇴장: 테이프 쪽을 축으로 2° 돌며 뜬다. 불투명도는 마지막 4f 에만 준다. */
export const peelOut = (frame: number, end: number, rest: number, dur = 8) => {
  const p = interpolate(frame, [end - dur, end], [0, 1], {...clamp, easing: EASE.exit});
  return {
    rotate: rest + 2 * p,
    translateY: -10 * p,
    elev: 1 + p,
    opacity: interpolate(frame, [end - 4, end], [1, 0], clamp),
  };
};

/** 역할 이름으로 쓰는 진입 진행도(0→1). 기존 tween() 과 같고 기본 곡선만 다르다. */
export const enterP = (frame: number, start: number, dur: number, role: 'enter' | 'enterText' | 'enterLarge' = 'enter') =>
  interpolate(frame, [start, start + Math.max(1, dur)], [0, 1], {...clamp, easing: EASE[role]});
```

## 3. `tokens.ts` — 하우스 재질과 광원 하나

```ts
// 하우스 재질(v2) — 한 영상 = 한 재질: 웜 잉크 책상 위의 크림 종이. 순백·순흑·#111 전면은 쓰지 않는다.
export const HOUSE = {
  paper: '#F5F2EA',       // 모든 그래픽의 바탕(= NOTE.paper)
  paperLight: '#FBF9F3',  // 종이 위에 덧붙인 쪽지·사진 테두리만. 전면 금지
  paperDeep: '#E9E4D8',   // 겹친 아랫장·괘선 종이(= NOTE.paperDeep)
  ink: '#26211E',         // 글자·선·배지(= NOTE.ink). 종이 위 대비 약 14:1
  inkSoft: 'rgba(38,33,30,0.72)',
  stage: '#1D1916',       // 책상(무대): 화자가 줄어들 때 뒤, 챕터 카드·반전 면
  rule: 'rgba(38,33,30,0.16)',
} as const;

/** 광원은 위 왼쪽 하나. 모든 그림자는 오른쪽 아래(dx : dy = 2 : 3), 색은 웜 잉크. */
export const LIGHT = {dx: 2, dy: 3, rgb: '38,33,30'} as const;

/** 높이 3단: 1 붙음(메모·자막 상자·테이프) · 2 살짝 뜸(사진·겹친 윗장) · 3 들림(진입·퇴장 중) */
export const ELEV = {
  1: {off: 3, blur: 6, a: 0.26, contact: 0.3},
  2: {off: 7, blur: 16, a: 0.22, contact: 0.22},
  3: {off: 14, blur: 34, a: 0.18, contact: 0},
} as const;

const mixN = (a: number, b: number, t: number) => a + (b - a) * t;
/** level 은 1~3 사이 실수도 받는다(진입 중 3 → 1 로 내려앉는다). */
const elevAt = (level: number) => {
  const l = Math.max(1, Math.min(3, level));
  const lo = ELEV[Math.floor(l) as 1 | 2 | 3];
  const hi = ELEV[Math.ceil(l) as 1 | 2 | 3];
  const t = l - Math.floor(l);
  return {off: mixN(lo.off, hi.off, t), blur: mixN(lo.blur, hi.blur, t), a: mixN(lo.a, hi.a, t),
    contact: mixN(lo.contact, hi.contact, t)};
};

/** 사각 종이·사진·자막 상자의 box-shadow: 접촉 그림자(선명) + 주변 그림자(부드러움) 두 겹. */
export const paperShadow = (level: number): string => {
  const e = elevAt(level);
  const x = (LIGHT.dx / LIGHT.dy) * e.off;
  return `1px 1.5px 1px rgba(${LIGHT.rgb},${e.contact.toFixed(3)}), `
    + `${x.toFixed(1)}px ${e.off.toFixed(1)}px ${e.blur.toFixed(1)}px rgba(${LIGHT.rgb},${e.a.toFixed(3)})`;
};

/**
 * 찢긴 모양(SVG path)의 그림자: 같은 path 를 한 장 더 깔고 '고정 블러'를 준 뒤 위치·불투명도만 움직인다.
 * drop-shadow 의 블러 값을 프레임마다 바꾸지 않는다(렌더 속도 규칙).
 */
export const paperShadowCopy = (level: number): {dx: number; dy: number; opacity: number; blur: number} => {
  const e = elevAt(level);
  return {dx: (LIGHT.dx / LIGHT.dy) * e.off, dy: e.off, opacity: e.a + e.contact * 0.4, blur: 8};
};

/** 종이 물성 — 0°·완전한 직선·완벽한 정렬은 디지털의 표식이다. 값은 시드로 고정한다. */
export const PAPER_MAT = {
  rotate: {note: [0.6, 1.8], photo: [1, 3], tape: [3, 8], sticker: [2, 4], speaker: [0.3, 0.6]}, // ±도, 0 금지
  jitter: 4,                           // 위치 ±px(그리드 선에서)
  cutEdge: {amp: 0.8, step: 55},       // 가위로 자른 변
  tornEdge: {amp: 6, step: 9},         // 찢긴 변(현행 tornTopPath)
  cornerNick: [1, 2],                  // 모서리 깎임 px
  grainTile: 256,                      // 질감 타일은 화면 px 고정 — 요소 scale 과 같이 줄이지 않는다
  halftone: {gap: 5.5, angle: 15},     // 사진·어두운 면에만
  misregister: {x: 2, y: 1},           // 시그널 색면이 잉크 선에서 어긋나는 양
  photo: {saturate: 0.75, border: [10, 14]},
} as const;

/** 시드 → 도착 각(도). 부호도 시드가 정한다. */
export const paperRotate = (seed: number, kind: keyof typeof PAPER_MAT.rotate): number => {
  const [lo, hi] = PAPER_MAT.rotate[kind];
  const u = Math.abs(Math.sin(seed * 12.9898 + 78.233) * 43758.5453) % 1;
  const sign = Math.sin(seed * 3.1) >= 0 ? 1 : -1;
  return sign * (lo + (hi - lo) * u);
};
```
